SELECT COUNT(birth_date)
FROM patients
WHERE YEAR(birth_date) >= (SELECT YEAR(GETDATE()) - 18);

SELECT COUNT(patient_weight)
FROM patients
WHERE (gender = 'M') AND patient_weight < (SELECT AVG(patient_weight) FROM patients WHERE gender = 'F');

SELECT attending_physician_id, COUNT(*) AS current_admissions
FROM admissions
WHERE discharge_date IS NULL
GROUP BY attending_physician_id

PRINT 'GROUP 2 SELECT A'
GO

SELECT nursing_unit_id, COUNT(*) AS current_admissions
FROM admissions
WHERE discharge_date IS NULL AND (SUBSTRING(nursing_unit_id, 1, 1) = '1' OR SUBSTRING(nursing_unit_id, 1, 1) = '2')
GROUP BY nursing_unit_id

GO

SELECT pa.patient_id, pa.first_name, pa.last_name, ad.room, ad.bed
FROM patients AS pa
JOIN admissions AS ad
ON pa.patient_id = ad.patient_id
WHERE ad.nursing_unit_id = '2SOUTH' AND ad.discharge_date IS NULL
ORDER BY pa.last_name

PRINT 'GROUP 3 SELECT B'
GO

SELECT ud.pharmacist_initials, ud.entered_date, ud.dosage
FROM unit_dose_orders AS ud
JOIN medications AS med
ON ud.medication_id = med.medication_id
WHERE med.medication_description = 'Morphine'
ORDER BY ud.pharmacist_initials

PRINT 'GROUP 4 SELECT A'
GO

SELECT py.physician_id, py.first_name, py.last_name, py.specialty
FROM physicians AS py
JOIN encounters AS en
ON py.physician_id = en.physician_id
JOIN patients AS pa
ON en.patient_id = pa.patient_id
WHERE pa.first_name = 'Walter' AND pa.last_name = 'Mitty'

GO

SELECT ad.patient_id, ad.room, ad.bed, med.medication_description
FROM admissions AS ad
JOIN unit_dose_orders AS ud
ON ad.patient_id = ud.patient_id
JOIN medications AS med
ON med.medication_id = ud.medication_id
WHERE ad.nursing_unit_id = 'CCU' AND ad.discharge_date IS NULL
