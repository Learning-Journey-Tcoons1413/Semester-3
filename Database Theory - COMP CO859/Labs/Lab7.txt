-- Create plumbing_services table
CREATE TABLE plumbing_services (
  service_id INT PRIMARY KEY, 
  service_description VARCHAR(30), 
  service_type CHAR(1) CHECK (service_type IN ('R', 'I', 'E')), 
  hourly_rate MONEY,
  sales_ytd MONEY); 

-- Create sales table
CREATE TABLE sales (
	sales_id INT PRIMARY KEY, 
	sales_date DATE, 
	amount MONEY, 
	service_id INT FOREIGN KEY REFERENCES plumbing_services(service_id));

-- Create View
CREATE VIEW high_end_services
AS
SELECT SUBSTRING(service_description, 1, 15) AS descriptions, sales_ytd
FROM plumbing_services
WHERE hourly_rate > (SELECT AVG(hourly_rate) FROM plumbing_services);

CREATE TRIGGER sales_insert
	ON sales
	AFTER INSERT
AS
BEGIN
	UPDATE ps
	SET ps.sales_ytd = ps.sales_ytd + i.amount
	FROM plumbing_services AS ps
	JOIN inserted AS i
	ON i.service_id = ps.service_id
END;
GO

CREATE TRIGGER sales_delete
	ON sales
	AFTER DELETE
AS
BEGIN
	UPDATE ps
	SET ps.sales_ytd = ps.sales_ytd - d.amount
	FROM plumbing_services AS ps
	JOIN deleted AS d
	ON ps.service_id = d.service_id
END;
GO

CREATE TRIGGER sales_update
	ON sales
	AFTER UPDATE
AS
BEGIN
	UPDATE ps
	SET ps.sales_ytd = ps.sales_ytd + (i.amount - d.amount)
	FROM plumbing_services AS ps
	JOIN deleted AS d
	ON ps.service_id = d.service_id
	JOIN inserted AS i
	ON ps.service_id = i.service_id
END;
GO
